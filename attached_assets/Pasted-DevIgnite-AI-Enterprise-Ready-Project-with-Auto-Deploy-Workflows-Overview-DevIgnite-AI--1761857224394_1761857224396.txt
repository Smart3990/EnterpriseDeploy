DevIgnite AI – Enterprise-Ready Project with Auto-Deploy Workflows

## 🚀 Overview
**DevIgnite AI** is a comprehensive AI-driven development and marketing suite engineered for zero-cost deployment, scalable automation, and enterprise-grade reliability. Inspired by Lovable’s simplicity, it integrates **AI generation, design tools, automation, analytics**, and **auto-deployment pipelines**.

---

## 🧩 Features
- **Unified Interface:** Lovable-inspired design, responsive, and modular.
- **AI Studio:** Generate text, images, voices, and videos with fallback providers (Groq → Gemini → HuggingFace → Ollama).
- **Automation Hub:** n8n integration (800+ free workflows) and Supabase Edge triggers.
- **Business Suite:** Analytics, invoices, client management.
- 🔥 **Lovable-Inspired UI** – Soft gradients, rounded cards, dark mode aesthetic.
- 💡 **AI Content Generator** – Text, image, voice, and video generation using free APIs.
- 📅 **Social Scheduler** – Post automation with fallback queue system.
- 🎨 **Design Studio** – Fabric.js-based canvas for live asset creation.
- ⚙️ **Automation Hub** – n8n + Supabase-driven visual workflows.
- 📊 **Analytics Dashboard** – Real-time Supabase data visualization.
- 🧠 **Multi-AI Fallback Engine** – Groq → Gemini → HuggingFace → Ollama.
- 🚀 **Auto Deploy** – One-click deploy to GitHub, Netlify, and Vercel.

### ⚡ DevIgnite Studio (Enhanced Layer)
- **AI Content Studio**: Text, Image, Voice, and Video generation.
- **Marketing Suite**: Social Media Scheduler (Buffer API free tier), AI asset library, and automation workflows.
- **Automation Engine**: Visual workflow builder (n8n + Supabase Edge), 800+ templates.
- **Business Tools**: Invoice generator, analytics dashboard, client portal.
- **Data Layer**: Supabase + PlanetScale + Pinecone (vector DB).
- **Auto Deploy**: GitHub, Netlify, Vercel, and Cloudflare Pages integration.

### 💡 AI Providers (Zero-Cost Multi-Provider Strategy)
| Type | Primary | Secondary | Fallbacks |
|------|----------|------------|------------|
| Text | Groq (Llama 3.1) | Gemini Flash | HuggingFace, Ollama |
| Image | Pollinations | Craiyon | HuggingFace SDXL |
| Voice | Coqui TTS | ElevenLabs | Play.ht |
| Video | Deforum | RunwayML (5s Free) | Colab Integration |

Each provider is automatically chained with fallback logic for continuous uptime.

/* ======= ContentGenerator ======= */
export function ContentGenerator(){
  const [mode, setMode] = useState<'text'|'image'|'voice'|'video'>('text');
  const [prompt, setPrompt] = useState('');
  const [loading, setLoading] = useState(false);
  const [result, setResult] = useState<any>(null);
  async function generate(){
    if(!prompt.trim()) return;
    setLoading(true);
    try{
      const chain = {
        text: ['/api/marketing/proxy/groq','/api/marketing/proxy/gemini','/api/marketing/proxy/huggingface'],
        image: ['/api/marketing/proxy/pollinations','/api/marketing/proxy/craiyon','/api/marketing/proxy/huggingface'],
        voice: ['/api/marketing/proxy/coqui','/api/marketing/proxy/elevenlabs'],
        video: ['/api/marketing/proxy/deforum']
      } as Record<string,string[]>;
      let output = null;
      for(const endpoint of chain[mode]){
        try{ const res = await fetch(endpoint, { method: 'POST', headers: {'Content-Type':'application/json'}, body: JSON.stringify({ prompt }) }); const j = await res.json(); if(j?.success || j?.url || j?.text){ output = j; break; } }catch(e){ continue; }
      }
      setResult(output);
    }catch(e){ console.error(e); }
    finally{ setLoading(false); }
  }
  return (
    <motion.div initial={{opacity:0,y:8}} animate={{opacity:1,y:0}} className="bg-gradient-to-br from-[#0f0f1f] to-[#1f1f3f] p-6 rounded-2xl shadow-lg border border-white/10 space-y-4">
      <h3 className="text-xl font-semibold text-white flex items-center gap-2"><Sparkles className="w-5 h-5 text-pink-400"/> DevIgnite AI Content</h3>
      <div className="flex flex-col md:flex-row gap-3">
        <select value={mode} onChange={e=>setMode(e.target.value as any)} className="p-3 rounded bg-white/10 text-white">
          <option value="text">Text</option>
          <option value="image">Image</option>
          <option value="voice">Voice</option>
          <option value="video">Video</option>
        </select>
        <input value={prompt} onChange={e=>setPrompt(e.target.value)} placeholder="Describe what you want to generate..." className="flex-1 p-3 rounded bg-white/10 text-white" />
        <button onClick={generate} disabled={loading} className="px-4 py-2 rounded bg-gradient-to-r from-purple-600 to-pink-600 text-white">{loading? 'Generating...':'Generate'}</button>
      </div>
      <div className="bg-black/30 p-4 rounded border border-white/10 min-h-[120px]">
        {result ? (
          <div className="space-y-2">
            {result.url && <img src={result.url} alt="generated" className="max-w-full rounded" />}
            {result.text && <pre className="text-white whitespace-pre-wrap">{result.text}</pre>}
          </div>
        ) : (
          <div className="text-gray-400">No results — try generating content.</div>
        )}
      </div>
    </motion.div>
  );
}

// 3) Error Boundary + Sentry integration stub
import * as Sentry from '@sentry/browser';
Sentry.init({ dsn: process.env.NEXT_PUBLIC_SENTRY_DSN || '' });

export class ErrorBoundary extends React.Component<any, { hasError: boolean }>{
  constructor(props:any){ super(props); this.state = { hasError:false }; }
  static getDerivedStateFromError(){ return { hasError:true }; }
  componentDidCatch(error:any, info:any){ Sentry.captureException(error); console.error(error, info); }
  render(){ if(this.state.hasError) return <div className="p-6 bg-red-900 text-white rounded">Something went wrong — contact support.</div>; return this.props.children; }
}

// 4) API client with circuit-breaker + retries + audit logging
async function auditLog(userId:string|null, action:string, meta:any){
  try{ await fetch('/api/enterprise/audit', { method:'POST', headers:{'Content-Type':'application/json'}, body: JSON.stringify({ userId, action, meta, ts: new Date().toISOString() }) }); }catch(e){ console.warn('Audit log failed'); }
}

export async function safeApi(endpoint:string, options:RequestInit={}){
  // simple retry + circuit-breaker placeholder
  const MAX_RETRY = 2; let attempt = 0; let lastErr:any = null;
  while(attempt <= MAX_RETRY){
    try{
      const res = await fetch(endpoint, options);
      if(!res.ok) throw new Error(`HTTP ${res.status}`);
      return await res.json();
    }catch(e){ lastErr = e; attempt++; const backoff = 200 * attempt; await new Promise(r=>setTimeout(r, backoff)); }
  }
  // after retries, record to audit/ops
  Sentry.captureMessage(`API failure: ${endpoint} - ${String(lastErr)}`);
  throw lastErr;
}

// 5) Rate limiter (client-side guard; server must enforce real limits)
const RATE_LIMIT_BUCKETS: Record<string, { tokens:number, last:number }> = {};
export function checkRate(key:string, limit=60){
  const now = Date.now(); const bucket = RATE_LIMIT_BUCKETS[key] || { tokens: limit, last: now };
  const elapsed = (now - bucket.last) / 1000; bucket.tokens = Math.min(limit, bucket.tokens + elapsed * (limit/60)); bucket.last = now;
  if(bucket.tokens < 1){ RATE_LIMIT_BUCKETS[key]=bucket; return false; }
  bucket.tokens -= 1; RATE_LIMIT_BUCKETS[key]=bucket; return true;
}

// 6) Preview sandbox orchestrator component (server-side must implement /api/preview/run with sandboxing)
export function PreviewRunner({ projectId, files }: { projectId?: string, files?: Record<string,string> }){
  const [status, setStatus] = useState('idle'); const [logs, setLogs] = useState<string[]>([]);
  async function run(){
    if(!checkRate('preview_run')){ alert('Rate limit reached — try later'); return; }
    setStatus('queued');
    try{
      const res = await safeApi('/api/preview/run', { method:'POST', headers:{'Content-Type':'application/json'}, body: JSON.stringify({ projectId, files }) });
      setStatus(res.status || 'running');
      setLogs(res.logs || []);
      auditLog(null, 'preview.run', { projectId });
    }catch(e){ setStatus('failed'); Sentry.captureException(e); }
  }
  return (
    <div className="space-y-2">
      <button onClick={run} className="px-4 py-2 rounded bg-indigo-600 text-white">Run Preview</button>
      <div className="bg-black/20 p-2 rounded min-h-[80px] font-mono text-sm text-white">{logs.map((l,i)=>(<div key={i}>{l}</div>))}</div>
      <div className="text-sm text-gray-300">Status: {status}</div>
    </div>
  );
}
